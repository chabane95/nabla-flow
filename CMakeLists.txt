cmake_minimum_required(VERSION 3.16)
project(nabla_flow LANGUAGES C)
add_executable(nablaflow src/main.c src/gpt.c src/data.c src/checkpoint.c src/recall.c src/scale.c)
target_compile_features(nablaflow PRIVATE c_std_11)
set_target_properties(nablaflow PROPERTIES C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(nablaflow PRIVATE /W4)
else()
    target_compile_options(nablaflow PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(nablaflow PRIVATE m)
endif()
enable_testing()
add_test(NAME math_and_pipeline COMMAND nablaflow check)
add_test(NAME full_dimensions COMMAND nablaflow scale info)
set_tests_properties(full_dimensions PROPERTIES PASS_REGULAR_EXPRESSION "50257,1024,768,12,12,124439808")
add_test(NAME recall_reference COMMAND nablaflow recall
    "${CMAKE_CURRENT_SOURCE_DIR}/results/model.nfg" "${CMAKE_CURRENT_SOURCE_DIR}/data/recall.tsv")
set_tests_properties(recall_reference PROPERTIES PASS_REGULAR_EXPRESSION "56,26,")
